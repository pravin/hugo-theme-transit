{{- /* Draws the home page map as an SVG. Each line (tag) is a row; each post is a station placed along x by its date. A post on several drawn lines gets a marker on each, joined by an interchange bar. Posts on none of the drawn lines sit on a grey "Other" line. */ -}} {{- $cfg := site.Params.transit | default dict }} {{- $data := partialCached "transit/lines.html" . "transit-lines" }} {{- $maxLines := $cfg.maxLines | default 8 | int }} {{- $labelCount := 12 }} {{- if isset $cfg "labels" }}{{ $labelCount = int $cfg.labels }}{{ end }} {{- $pages := (where site.RegularPages "Type" "in" site.MainSections).ByDate }} {{- with $pages }} {{- $lines := first $maxLines $data.lines }} {{- /* Posts not on any drawn line ride the "Other" line */}} {{- $onMap := slice }} {{- range $lines }}{{ $onMap = $onMap | append .urls }}{{ end }} {{- $others := slice }} {{- $otherURLs := slice }} {{- range $pages }} {{- if not (in $onMap .RelPermalink) }}{{ $others = $others | append . }}{{ $otherURLs = $otherURLs | append .RelPermalink }}{{ end }} {{- end }} {{- if $others }} {{- $lines = $lines | append (dict "name" ($cfg.otherName | default "Other topics") "code" "··" "colour" "" "url" "" "stations" $others "urls" $otherURLs) }} {{- end }} {{- /* Geometry, in viewBox units */}} {{- $xl := 170.0 }}{{ $xr := 1100.0 }} {{- $top := cond (gt $labelCount 0) 200.0 50.0 }} {{- $step := 46.0 }} {{- $width := 1320 }} {{- $height := add (add $top (mul $step (sub (len $lines) 1))) 80.0 }} {{- $min := (index $pages 0).Date.Unix }} {{- $max := (index $pages.Reverse 0).Date.Unix }} {{- $span := sub $max $min }} {{- $axis := sub $height 28.0 }} {{- /* x position of every post, keyed by URL */}} {{- $s := newScratch }} {{- range $pages }} {{- $s.SetInMap "x" .RelPermalink (partial "transit/x.html" (dict "t" .Date.Unix "min" $min "span" $span "xl" $xl "xr" $xr)) }} {{- end }} {{- $xs := $s.Get "x" }} {{- /* Name the newest stations, skipping any that would sit on top of a newer label */}} {{- $labelled := slice }} {{- $lastX := 100000.0 }} {{- range $pages.Reverse }} {{- if ge (len $labelled) $labelCount }}{{ break }}{{ end }} {{- $x := index $xs .RelPermalink }} {{- if ge (sub $lastX $x) 28 }}{{ $labelled = $labelled | append .RelPermalink }}{{ $lastX = $x }}{{ end }} {{- end }} {{- end }}
